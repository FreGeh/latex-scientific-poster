\RequirePackage{fix-cm}
\documentclass[UKenglish]{article}
\usepackage{poster}

% --- Poster settings ---------------------------------------------------------
% Paper presets: a0-landscape, a0-portrait, conference-wide, open (content-height draft).
\posterpaper{a0-landscape}
% Brand profiles: template, kit, hlrs. Comment out for an unbranded poster.
\useposterbrand{template}
% Use canvas for explicit rows or columns for automatic columns on wide paper.
\postersetup{flow=canvas}

\title{Data locality for repeated\texorpdfstring{\\}{ }scientific workflows}
\author{A. Researcher, B. Researcher, C. Researcher}
% Leave empty to omit the contact block.
\newcommand{\postercontactemail}{researcher@example.org}

% --- Typography overrides ---------------------------------------------------
\postersetup{title size=62bp,title leading=80bp,heading size=46bp}
\DeclareMathSizes{40bp}{40bp}{28bp}{20bp}

% --- Bibliography and figures -----------------------------------------------
\usepackage[backend=biber,style=numeric,sorting=none]{biblatex}
\addbibresource{references.bib}
\renewcommand*{\bibfont}{\postersmall}
\newtoggle{posterreferences}
\AtDataInput{\global\toggletrue{posterreferences}}
\defbibheading{posterreferences}{\noindent\textbf{References}\par\nobreak}
\input{figures.tex}

% --- Document layout --------------------------------------------------------
\begin{document}
\makeposterheader
\begin{postercontent}
  \input{content.tex}
\end{postercontent}

% Keep the bibliography outside the body so every content row precedes it.
\ifboolexpr{togl {posterreferences} or not test {\ifdefvoid{\postercontactemail}}}{%
  \posterspacer
  \vfill
  \begingroup
    \postersmall
    \setlength{\parskip}{0pt}
    \setlength{\bibitemsep}{\posterpargap}
    \newcommand{\contactblock}{%
      \raggedleft\textbf{Contact}\par\postercontactemail\par}
    \iftoggle{posterreferences}{%
      \ifdefvoid{\postercontactemail}
        {\printbibliography[heading=posterreferences]}
        {\begin{posterrow}{3,1}
           \postercolumn{\printbibliography[heading=posterreferences]}
           \postercolumn{\contactblock}
         \end{posterrow}}%
    }{\contactblock}%
  \endgroup
}{}
\end{document}
